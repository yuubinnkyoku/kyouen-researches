## 再現用独立検査

```python
from fractions import Fraction

bounds = [
    (111, 1415), (112, 1431), (113, 1445), (114, 1461),
    (115, 1477), (116, 1491), (117, 1507), (118, 1521),
    (119, 1537), (120, 1553),
]
for q, kp in bounds:
    r = q - 3
    a = Fraction(r + 2, r + 1)
    for k, exists in ((kp - 2, False), (kp, True)):
        y = Fraction(k, k + 1) * a
        H = (a - y) * y**k
        if exists:
            assert H > r - y, (q, k)
        else:
            assert H < r - a, (q, k)
print("20 exact rational checks passed")
```
