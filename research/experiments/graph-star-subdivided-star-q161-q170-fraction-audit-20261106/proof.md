## 再現コード

```python
from fractions import Fraction

for q in range(161, 171):
    r = q - 3
    k_plus = 2193 + 16 * (q - 161)
    for k in (k_plus - 2, k_plus):
        a = Fraction(r + 2, r + 1)
        y = Fraction(k, k + 1) * a
        h = (a - y) * y**k
        if k == k_plus - 2:
            assert h < r - a, (q, k)
        else:
            assert 1 < y < a, (q, k)
            assert h > r - y, (q, k)
    print(q, "PASS")
```
